cmake_minimum_required(VERSION 3.28)
project(recoverx_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(recoverx_core STATIC src/signatures.cpp)
set_target_properties(recoverx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(recoverx_core PUBLIC include)
if(MSVC)
  target_compile_options(recoverx_core PRIVATE /W4 /WX)
else()
  target_compile_options(recoverx_core PRIVATE -Wall -Wextra -Werror)
endif()
option(RECOVERX_PYTHON "Build optional pybind11 extension" OFF)
if(RECOVERX_PYTHON)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_recoverx_native bindings/module.cpp)
  target_link_libraries(_recoverx_native PRIVATE recoverx_core)
  install(TARGETS _recoverx_native LIBRARY DESTINATION . RUNTIME DESTINATION .)
endif()
enable_testing()
add_executable(signatures_test tests/signatures_test.cpp)
target_link_libraries(signatures_test PRIVATE recoverx_core)
if(MSVC)
  target_compile_options(signatures_test PRIVATE /UNDEBUG)
else()
  target_compile_options(signatures_test PRIVATE -UNDEBUG)
endif()
add_test(NAME signature_bounds_and_overlap COMMAND signatures_test)
